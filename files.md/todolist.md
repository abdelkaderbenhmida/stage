What each role must do (tell the model)
Role: docker
# roles/docker/tasks/main.yml
# - Install Docker CE on Ubuntu 22.04
# - Add Docker's official GPG key + apt repo
# - Install packages: docker-ce, docker-ce-cli, containerd.io
# - Enable + start docker.service
# - Add user 'devops' to the docker group
# - Configure containerd for Kubernetes:
#     - run: containerd config default > /etc/containerd/config.toml
#     - set SystemdCgroup = true in /etc/containerd/config.toml
#     - restart containerd
# - Do NOT install Docker Compose (not needed)
# - Note: kubeadm 1.28+ uses containerd, not dockerd. 
#   Docker is installed but the CRI that K8s uses is containerd.
Role: k8s_common
# roles/k8s_common/tasks/main.yml
# Run on ALL nodes (master + workers)
#
# - Disable swap:
#     - swapoff -a
#     - comment out swap lines in /etc/fstab (use replace module)
# - Load kernel modules:
#     - modprobe overlay
#     - modprobe br_netfilter
#     - make modules persistent in /etc/modules-load.d/k8s.conf
# - Set sysctl for Kubernetes:
#     - net.bridge.bridge-nf-call-iptables  = 1
#     - net.bridge.bridge-nf-call-ip6tables = 1
#     - net.ipv4.ip_forward                = 1
#     - apply with sysctl --system
# - Add Kubernetes apt repo:
#     - install apt-transport-https, ca-certificates, curl
#     - add Google Cloud GPG key for k8s
#     - add repo: deb https://pkgs.k8s.io/core:/stable:/v1.28/deb/ /
# - Install packages (pin to v1.28.x):
#     - kubelet
#     - kubeadm
#     - kubectl
# - Hold packages so they don't auto-upgrade (apt hold)
# - Enable kubelet service
Role: k8s_master
# roles/k8s_master/tasks/main.yml
# Run on master only (when: inventory_hostname == master_name)
#
# - Run kubeadm init:
#     kubeadm init --apiserver-advertise-address=<master_ip>
#                  --pod-network-cidr=192.168.0.0/16
#                  --service-cidr=10.96.0.0/12
# - Create kubeconfig for devops user:
#     - mkdir -p /home/devops/.kube
#     - copy /etc/kubernetes/admin.conf to /home/devops/.kube/config
#     - chown devops:devops
# - Install Calico CNI:
#     - kubectl apply -f https://raw.githubusercontent.com/projectcalico/calico/v3.26.1/manifests/tigera-operator.yaml
#     - kubectl apply -f https://raw.githubusercontent.com/projectcalico/calico/v3.26.1/manifests/custom-resources.yaml
#   OR use the Calico manifest (calico.yaml) — model can pick
# - Generate join command:
#     kubeadm token create --print-join-command
# - Store output in a variable
# - Add the join command to a host fact OR write it to /tmp/join-command.sh
#   so the worker role can fetch it
Role: k8s_worker
# roles/k8s_worker/tasks/main.yml
# Run on workers only
#
# - Get the join command (either from a fact set by master, 
#   or by running kubeadm token create on master via delegate_to)
# - Run: <join command from kubeadm token create --print-join-command>
# - Wait for node to be Ready:
#     - delegate_to: master
#     - kubectl get nodes
#     - retry until worker status is Ready
Role: k8s_reset (OPTIONAL but save headaches later)
# roles/k8s_reset/tasks/main.yml
# Useful to re-run ansible-playbook without errors if cluster already exists
# - Run on all nodes:
#     kubeadm reset -f
#     rm -rf /etc/kubernetes,/var/lib/etcd,/var/lib/kubelet,/var/lib/cni
#     rm -rf /home/devops/.kube
#     iptables -F && iptables -t nat -F && iptables -t mangle -F
#     ipvsadm --clear   (if ipvsadm installed)
# - Do NOT run automatically — only on demand
