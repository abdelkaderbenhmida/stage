From master node:
# 1. Pods + Services + HPA status
kubectl get pods,svc,hpa -n devops-platform

# 2. Internal service health
for svc in users products orders; do
  kubectl run --rm -it curl-$svc --image=curlimages/curl --restart=Never --timeout=10s -- \
    curl -s http://${svc}-service.devops-platform.svc.cluster.local/health
done

# 3. Data endpoints
for svc in users products orders; do
  kubectl run --rm -it curl-$svc --image=curlimages/curl --restart=Never --timeout=10s -- \
    curl -s http://${svc}-service.devops-platform.svc.cluster.local/${svc}
done

# 4. Metrics
kubectl run --rm -it curl-metrics --image=curlimages/curl --restart=Never --timeout=10s -- \
  curl -s http://users-service.devops-platform.svc.cluster.local/metrics | head -10

# 5. Self-healing (replace pod name)
kubectl delete pod -l app=users-service -n devops-platform --grace-period=0 --force
kubectl get pods -n devops-platform -w
